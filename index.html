Panduan Persediaan Google Apps Script
1
Buka Google Sheets anda
Buat spreadsheet baru atau guna yang sedia ada. Pastikan ada 3 tab/sheet: DV3, PPM, dan Kontraktor. Row pertama adalah header.
2
Buka Apps Script Editor
Dalam Google Sheets, klik Extensions → Apps Script. Padam kod yang ada dan tampal kod berikut:
// ═══════════════════════════════════════════════════════
// JPPF Dashboard — Google Apps Script Backend
// Klik untuk SALIN kod penuh
// ═══════════════════════════════════════════════════════

const SPREADSHEET_ID = '1oPbeDl3kw_0Wo2QMLj_uspYuIi1U8FVgLWzPueZJ4mQ';

function doGet(e) {
  const sheet = e.parameter.sheet || 'DV3';
  const action = e.parameter.action || 'read';
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const ws = ss.getSheetByName(sheet);
  if (!ws) return json({error: 'Sheet tidak wujud: ' + sheet});

  if (action === 'read') {
    const data = ws.getDataRange().getValues();
    const headers = data[0];
    const rows = data.slice(1).map((r,i) => {
      const obj = {_row: i + 2};
      headers.forEach((h,j) => obj[h] = r[j]);
      return obj;
    });
    return json({success:true, data:rows});
  }
  return json({error:'Action tidak dikenali'});
}

function doPost(e) {
  const payload = JSON.parse(e.postData.contents);
  const {sheet, action, data, rowNum} = payload;
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const ws = ss.getSheetByName(sheet);
  if (!ws) return json({error:'Sheet tidak wujud'});

  if (action === 'append') {
    const headers = ws.getRange(1, 1, 1, ws.getLastColumn()).getValues()[0];
    const row = headers.map(h => data[h] !== undefined ? data[h] : '');
    ws.appendRow(row);
    const newRow = ws.getLastRow();
    return json({success:true, rowNum:newRow});
  }

  if (action === 'update') {
    const headers = ws.getRange(1, 1, 1, ws.getLastColumn()).getValues()[0];
    const row = headers.map(h => data[h] !== undefined ? data[h] : '');
    ws.getRange(rowNum, 1, 1, row.length).setValues([row]);
    return json({success:true});
  }

  if (action === 'delete') {
    ws.deleteRow(rowNum);
    return json({success:true});
  }

  return json({error:'Action tidak dikenali'});
}

function json(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
3
Deploy sebagai Web App
Klik Deploy → New Deployment. Pilih type: Web App. Set "Execute as": Me, "Who has access": Anyone. Klik Deploy dan salin URL yang diberi.
4
Masukkan URL di atas
Tampal URL Web App ke dalam kotak URL untuk DV3, PPM, atau Kontraktor di bahagian konfigurasi atas. Tambah parameter ?sheet=DV3 (atau PPM/Kontraktor) di hujung URL.
5
Format Header Google Sheets
Pastikan baris pertama setiap sheet mengandungi header yang betul:
DV3: Tahun | Penyenggara | TakSiap | Siap
PPM: Bulan | Tahun | Penyenggara | TakSiap | Siap
Kontraktor: Bulan | Tahun | Kontraktor | Zon | JumlahDoket | JumlahRM
